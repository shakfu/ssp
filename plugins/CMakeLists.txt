cmake_minimum_required(VERSION 3.15)
project(plugins)

add_subdirectory(sfct)
add_subdirectory(rack)
add_subdirectory(radio)
add_subdirectory(glitch)
add_subdirectory(edrums)
add_subdirectory(pstretch)
add_subdirectory(bard)
add_subdirectory(chorus)
add_subdirectory(dexed)
add_subdirectory(gseq)

# teletype builds its firmware from sources `make deps` fetches (scripts/fetch_teletype.sh)
if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/src/teletype/monome-rack")
    add_subdirectory(teletype)
else ()
    message(STATUS "skipping teletype: run `make deps`")
endif ()

# csound, chuck, faust and scsynth link prebuilt libraries: `make deps` builds them
if (TARGET_SSP)
    set(DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/ssp")
else ()
    set(DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/host")
endif ()
foreach (P csound chuck faust scsynth)
    if (EXISTS "${DEPS_DIR}/lib/lib${P}.a")
        add_subdirectory(${P})
    else ()
        message(STATUS "skipping ${P}: run `make deps`")
    endif ()
endforeach ()

# sapf: the sapf fork at the commit scripts/fetch_sapf.sh pins, or SAPF_DIR; and the deps' libsndfile
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/fetch_sapf.sh" SAPF_REF_LINE REGEX "^SAPF_REF=")
string(REGEX REPLACE "^SAPF_REF=([0-9a-f]+).*" "\\1" SAPF_REF "${SAPF_REF_LINE}")
if (NOT SAPF_DIR)
    set(SAPF_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/src/sapf-${SAPF_REF}")
endif ()
if (EXISTS "${SAPF_DIR}/src/engine/CMakeLists.txt" AND EXISTS "${DEPS_DIR}/lib/libsndfile.a")
    add_subdirectory(sapf)
else ()
    message(STATUS "skipping sapf: run `make deps`")
endif ()

# alda: psnd's Alda core at the commit scripts/fetch_alda.sh pins, or ALDA_DIR
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/fetch_alda.sh" ALDA_REF_LINE REGEX "^ALDA_REF=")
string(REGEX REPLACE "^ALDA_REF=([0-9a-f]+).*" "\\1" ALDA_REF "${ALDA_REF_LINE}")
if (NOT ALDA_DIR)
    set(ALDA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/src/alda-${ALDA_REF}")
endif ()
if (EXISTS "${ALDA_DIR}/impl/interpreter.c")
    add_subdirectory(alda)
else ()
    message(STATUS "skipping alda: run `make deps`")
endif ()

# breaks: _core at the commit scripts/fetch_core.sh pins, with its engine generated for this target
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/fetch_core.sh" CORE_REF_LINE REGEX "^CORE_REF=")
string(REGEX REPLACE "^CORE_REF=([0-9a-f]+).*" "\\1" CORE_REF "${CORE_REF_LINE}")
set(CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/src/core-${CORE_REF}")
if (TARGET_SSP)
    set(CORE_TARGET ssp)
else ()
    set(CORE_TARGET host)
endif ()
if (EXISTS "${CORE_DIR}/gen/${CORE_TARGET}/engine.c")
    add_subdirectory(breaks)
else ()
    message(STATUS "skipping breaks: run `make deps`")
endif ()

# fsyn: FluidSynth, which `make deps` builds (scripts/build_fluidsynth.sh)
if (EXISTS "${DEPS_DIR}/lib/libfluidsynth.a")
    add_subdirectory(fsyn)
else ()
    message(STATUS "skipping fsyn: run `make deps`")
endif ()
