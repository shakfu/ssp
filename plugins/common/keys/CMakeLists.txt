# A plugin's grammar: compiles external/tree-sitter-<grammar>/src/parser.c into `target`, and writes its
# queries/highlights.scm to tree-sitter-<grammar>-highlights.h as <GRAMMAR>_HIGHLIGHTS
set(SSP_EXTERNAL_DIR ${CMAKE_CURRENT_LIST_DIR}/../../../external)
function(ssp_highlights target grammar)
    set(dir "${SSP_EXTERNAL_DIR}/tree-sitter-${grammar}")
    string(TOUPPER ${grammar} name)
    file(READ "${dir}/queries/highlights.scm" query)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tree-sitter-${grammar}-highlights.h"
            "// generated by ssp_highlights() from external/tree-sitter-${grammar}/queries/highlights.scm\n"
            "static const char* const ${name}_HIGHLIGHTS = R\"scm(${query})scm\";\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dir}/queries/highlights.scm")
    target_sources(${target} PRIVATE "${dir}/src/parser.c")
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
endfunction()
